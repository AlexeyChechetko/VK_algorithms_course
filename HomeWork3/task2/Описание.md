# Пояснения

```go
func copyTime(n, x, y int64) int64 {
	// Если копии не нужны, время равно 0.
	if n <= 0 {
		return 0
	}
	// Время изготовления одной копии должно быть положительным.
	if x <= 0 || y <= 0 {
		return -1
	}
	// Изначально есть только один оригинал, поэтому сначала работает
	// только самый быстрый ксерокс. Получаем первую копию.
	first := min(x, y)
	// Теперь есть два листа, и можно использовать оба ксерокса.
	remaining := n - 1
	// Ищем дополнительное время. Нижняя граница — 0.
	// Верхней границы хватит даже при работе только быстрого ксерокса.
	left, right := int64(0), remaining*first
	for left < right {
		middle := (left + right) / 2
		// Целочисленное деление показывает число готовых копий
		// у каждого ксерокса за middle единиц времени.
		if middle/x+middle/y >= remaining {
			// Времени достаточно. Проверяем, можно ли успеть быстрее.
			// Само middle тоже может быть минимальным ответом.
			right = middle
		} else {
			// Времени не хватает, поэтому ответ строго больше middle.
			left = middle + 1
		}
	}
	// left == right — минимальное время после изготовления первой копии.
	return first + left
}
```

Алгоритм работает за `O(log((n-1)*min(x,y)))`.
